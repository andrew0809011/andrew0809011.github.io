import { useState, useEffect } from 'react'

export const PrivateLock = () => {
  const [unlocked, setUnlocked] = useState(false)
  const [typed, setTyped] = useState('')
  const [err, setErr] = useState(false)

  useEffect(() => {
    if (typeof window === 'undefined') return

    if (localStorage.getItem('__author_kc') === '1') {
      setUnlocked(true)
      return
    }

    const checkHash = () => {
      if (window.location.hash === '#unlock') {
        localStorage.setItem('__author_kc', '1')
        setUnlocked(true)
      }
    }
    checkHash()
    window.addEventListener('hashchange', checkHash)

    window.unlock = () => {
      localStorage.setItem('__author_kc', '1')
      setUnlocked(true)
    }

    const KONAMI = ['ArrowUp','ArrowUp','ArrowDown','ArrowDown','ArrowLeft','ArrowRight','ArrowLeft','ArrowRight']
    let seq = []
    const onKey = (e) => {
      seq = [...seq.slice(-(KONAMI.length - 1)), e.key]
      if (seq.join(',') === KONAMI.join(',')) {
        localStorage.setItem('__author_kc', '1')
        setUnlocked(true)
      }
    }
    window.addEventListener('keydown', onKey)

    return () => {
      window.removeEventListener('hashchange', checkHash)
      window.removeEventListener('keydown', onKey)
    }
  }, [])

  const tryPassword = () => {
    if (typed === 'andrew') {
      localStorage.setItem('__author_kc', '1')
      setUnlocked(true)
    } else {
      setErr(true)
      setTyped('')
      setTimeout(() => setErr(false), 1500)
    }
  }

  if (unlocked) return null

  return (
    <div style={{
      position: 'fixed', top: 0, left: 0, right: 0, bottom: 0,
      zIndex: 9999,
      display: 'flex', flexDirection: 'column', alignItems: 'center', justifyContent: 'center',
      gap: '28px',
      background: 'linear-gradient(135deg, #0a0f1e 0%, #1a1040 50%, #0a0f1e 100%)',
      fontFamily: 'ui-monospace, SFMono-Regular, Menlo, monospace',
    }}>
      <div style={{ fontSize: '56px', lineHeight: 1 }}>🔒</div>

      <div style={{ textAlign: 'center', lineHeight: 1.5 }}>
        <div style={{ fontSize: '20px', fontWeight: 700, color: '#c7d2fe', letterSpacing: '0.04em' }}>
          Private Article
        </div>
        <div style={{ fontSize: '11px', color: '#4f46e5', letterSpacing: '0.2em', textTransform: 'uppercase', marginTop: '4px' }}>
          Author Access Only
        </div>
      </div>

      <div style={{ display: 'flex', gap: '8px', flexDirection: 'column', alignItems: 'center' }}>
        <div style={{ display: 'flex', gap: '8px' }}>
          <input
            type="password"
            placeholder="passphrase…"
            value={typed}
            autoFocus
            onChange={e => setTyped(e.target.value)}
            onKeyDown={e => e.key === 'Enter' && tryPassword()}
            style={{
              background: err ? 'rgba(239,68,68,0.1)' : 'rgba(99,102,241,0.08)',
              border: err ? '1px solid rgba(239,68,68,0.5)' : '1px solid rgba(99,102,241,0.35)',
              borderRadius: '8px',
              padding: '10px 16px',
              color: '#e0e7ff',
              fontSize: '14px',
              outline: 'none',
              width: '200px',
              letterSpacing: '0.08em',
              transition: 'border-color 0.2s, background 0.2s',
            }}
          />
          <button
            onClick={tryPassword}
            style={{
              background: 'rgba(99,102,241,0.7)',
              border: 'none',
              borderRadius: '8px',
              padding: '10px 18px',
              color: '#fff',
              fontSize: '16px',
              cursor: 'pointer',
              transition: 'background 0.2s',
            }}
          >→</button>
        </div>
        <div style={{ fontSize: '11px', color: err ? '#f87171' : 'transparent', transition: 'color 0.2s', height: '16px' }}>
          incorrect passphrase
        </div>
      </div>

      <div style={{ color: '#1e293b', fontSize: '10px', letterSpacing: '0.08em' }}>· · ·</div>
    </div>
  )
}
